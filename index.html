/**************************************************** * 醫奈所會議室預約系統 * Google Apps Script 後端 * * 修正版 * * 修正： * 1. Google Sheet 時間欄位 Date 物件造成 1899/12/30 * 2. 前端出現 NaN:00 * 3. 查詢預約時間格式統一為 HH:mm ****************************************************/ /* ================================================== 基本設定 ================================================== */ const BOOKING_SHEET_NAME = '預約資料'; const CANCEL_LOG_SHEET_NAME = '取消紀錄'; const BOOKING_HEADERS = [ 'room', 'date', 'time', 'name', 'ext', 'bookingId', 'cancelCode', '管理操作' ]; const CANCEL_LOG_HEADERS = [ '取消時間', '會議室', '日期', '時間', '預約人', '分機', 'bookingId', '取消方式', '操作者' ]; /* ================================================== 取得預約工作表 ================================================== */ function getBookingSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName( BOOKING_SHEET_NAME ); if (!sheet) { sheet = ss.insertSheet( BOOKING_SHEET_NAME ); } ensureBookingHeaders(sheet); return sheet; } /* ================================================== 建立 / 修正標題 ================================================== */ function ensureBookingHeaders(sheet) { const requiredColumns = BOOKING_HEADERS.length; if ( sheet.getMaxColumns() < requiredColumns ) { sheet.insertColumnsAfter( sheet.getMaxColumns(), requiredColumns - sheet.getMaxColumns() ); } sheet .getRange( 1, 1, 1, requiredColumns ) .setValues([ BOOKING_HEADERS ]); sheet .getRange( 1, 1, 1, requiredColumns ) .setFontWeight('bold') .setBackground('#2b6cb0') .setFontColor('white'); sheet.setFrozenRows(1); /* 管理操作欄設定下拉選單 */ const lastRows = Math.max( sheet.getMaxRows() - 1, 1 ); const rule = SpreadsheetApp .newDataValidation() .requireValueInList( ['取消'], true ) .setAllowInvalid(true) .build(); sheet .getRange( 2, 8, lastRows, 1 ) .setDataValidation(rule); /* * C 欄時間統一使用文字格式 * * 避免 Google Sheet 將 12:00 * 自動當成 Date 物件。 */ sheet .getRange( 2, 3, Math.max( sheet.getMaxRows() - 1, 1 ), 1 ) .setNumberFormat('@'); } /* ================================================== 取得取消紀錄工作表 ================================================== */ function getCancelLogSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName( CANCEL_LOG_SHEET_NAME ); if (!sheet) { sheet = ss.insertSheet( CANCEL_LOG_SHEET_NAME ); } sheet .getRange( 1, 1, 1, CANCEL_LOG_HEADERS.length ) .setValues([ CANCEL_LOG_HEADERS ]); sheet .getRange( 1, 1, 1, CANCEL_LOG_HEADERS.length ) .setFontWeight('bold') .setBackground('#718096') .setFontColor('white'); sheet.setFrozenRows(1); return sheet; } /* ================================================== 網頁 GET ================================================== */ function doGet() { const sheet = getBookingSheet(); const lastRow = sheet.getLastRow(); if (lastRow <= 1) { return jsonResponse([]); } const values = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); const result = []; values.forEach(row => { if (!row[0]) return; result.push({ room: String(row[0]), date: normalizeDate(row[1]), /* * ★ 重要修正 * * 原本： * String(row[2]) * * 現在： * normalizeTime(row[2]) */ time: normalizeTime(row[2]), name: String(row[3]), ext: String(row[4]), bookingId: String(row[5] || '') }); }); /* * 不回傳 cancelCode * * 避免其他使用者從月曆資料 * 取得別人的取消碼。 */ return jsonResponse( result ); } /* ================================================== POST API ================================================== */ function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(15000); const data = JSON.parse( e.postData.contents ); const action = String( data.action || '' ); switch (action) { case 'book': return createBooking(data); case 'find': return findBooking(data); case 'cancel': return cancelBooking(data); default: return jsonResponse({ success: false, message: '未知的操作' }); } } catch (error) { console.error(error); return jsonResponse({ success: false, message: '系統錯誤：' + error.toString() }); } finally { try { lock.releaseLock(); } catch (error) {} } } /* ================================================== 新增預約 ================================================== */ function createBooking(data) { const room = String(data.room || '').trim(); const date = String(data.date || '').trim(); const times = Array.isArray(data.times) ? data.times.map( x => normalizeTime(x) ) : []; const name = String(data.name || '').trim(); const ext = String(data.ext || '').trim(); if ( !room || !date || times.length === 0 || !name || !ext ) { return jsonResponse({ success: false, message: '預約資料不完整' }); } /* * 去除重複時段 */ const uniqueTimes = [...new Set(times)] .sort(); /* * 確認時間格式 */ const validTimes = uniqueTimes.every( time => /^\d{2}:00$/.test(time) ); if (!validTimes) { return jsonResponse({ success: false, message: '預約時段格式錯誤' }); } const sheet = getBookingSheet(); const lastRow = sheet.getLastRow(); const values = lastRow > 1 ? sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues() : []; /* * 防止重複預約 */ for ( const time of uniqueTimes ) { const occupied = values.some(row => { return ( String(row[0]) === room && normalizeDate(row[1]) === date && normalizeTime(row[2]) === time ); }); if (occupied) { return jsonResponse({ success: false, message: `${time} 已經被預約。請重新選擇連續時段。` }); } } /* * 建立唯一預約 ID */ const bookingId = Utilities.getUuid(); /* * 建立 6 碼取消碼 */ const cancelCode = generateCancelCode(); /* * 寫入 Google Sheet */ uniqueTimes.forEach(time => { sheet.appendRow([ room, date, time, name, ext, bookingId, cancelCode, '' ]); }); /* * 回傳成功資訊 */ return jsonResponse({ success: true, message: '預約成功', bookingId: bookingId, cancelCode: cancelCode, room: room, date: date, times: uniqueTimes, name: name, ext: ext }); } /* ================================================== 查詢取消碼 ================================================== */ function findBooking(data) { const cancelCode = String( data.cancelCode || '' ) .trim() .toUpperCase(); if (!cancelCode) { return jsonResponse({ success: false, message: '請輸入取消碼' }); } const sheet = getBookingSheet(); const lastRow = sheet.getLastRow(); if (lastRow <= 1) { return jsonResponse({ success: false, message: '目前沒有預約資料' }); } const values = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); const bookings = {}; values.forEach(row => { const rowCancelCode = String( row[6] || '' ) .trim() .toUpperCase(); if ( rowCancelCode !== cancelCode ) { return; } const bookingId = String( row[5] || '' ); if (!bookingId) return; if ( !bookings[bookingId] ) { bookings[bookingId] = { bookingId: bookingId, room: String(row[0]), date: normalizeDate(row[1]), name: String(row[3]), ext: String(row[4]), times: [] }; } /* * ★ 重要修正 * * 原本： * String(row[2]) * * 現在： * normalizeTime(row[2]) */ bookings[bookingId] .times .push( normalizeTime(row[2]) ); }); const result = Object.values( bookings ); if ( result.length === 0 ) { return jsonResponse({ success: false, message: '找不到符合的預約，請確認取消碼。' }); } result.forEach(booking => { booking.times.sort(); }); return jsonResponse({ success: true, bookings: result }); } /* ================================================== 使用者取消預約 ================================================== */ function cancelBooking(data) { const bookingId = String( data.bookingId || '' ).trim(); const cancelCode = String( data.cancelCode || '' ) .trim() .toUpperCase(); if ( !bookingId || !cancelCode ) { return jsonResponse({ success: false, message: '取消資料不完整' }); } const sheet = getBookingSheet(); const lastRow = sheet.getLastRow(); if (lastRow <= 1) { return jsonResponse({ success: false, message: '找不到預約' }); } const values = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); const rowsToDelete = []; const bookingInfo = { room: '', date: '', name: '', ext: '', times: [] }; values.forEach( (row, index) => { const rowBookingId = String( row[5] || '' ); const rowCancelCode = String( row[6] || '' ) .trim() .toUpperCase(); if ( rowBookingId === bookingId && rowCancelCode === cancelCode ) { rowsToDelete.push( index + 2 ); bookingInfo.room = String(row[0]); bookingInfo.date = normalizeDate(row[1]); bookingInfo.name = String(row[3]); bookingInfo.ext = String(row[4]); /* * ★ 時間格式修正 */ bookingInfo.times.push( normalizeTime(row[2]) ); } } ); if ( rowsToDelete.length === 0 ) { return jsonResponse({ success: false, message: '取消碼錯誤，無法取消此預約。' }); } /* * 寫入取消紀錄 */ writeCancelLog({ room: bookingInfo.room, date: bookingInfo.date, times: bookingInfo.times, name: bookingInfo.name, ext: bookingInfo.ext, bookingId: bookingId, method: '使用者取消', operator: bookingInfo.name }); /* * 由後往前刪除 */ rowsToDelete .sort( (a, b) => b - a ) .forEach( rowNumber => { sheet.deleteRow( rowNumber ); } ); return jsonResponse({ success: true, message: '預約已取消', count: rowsToDelete.length }); } /* ================================================== 管理員直接從 Google Sheet 取消 ================================================== */ function onEdit(e) { try { if (!e || !e.range) return; const range = e.range; const sheet = range.getSheet(); /* * 只處理預約資料工作表 */ if ( sheet.getName() !== BOOKING_SHEET_NAME ) { return; } /* * 只處理 H 欄 */ if ( range.getColumn() !== 8 ) { return; } /* * 只處理資料列 */ if ( range.getRow() <= 1 ) { return; } const value = String( range.getValue() || '' ) .trim(); if ( value !== '取消' ) { return; } const row = range.getRow(); const bookingId = String( sheet .getRange( row, 6 ) .getValue() || '' ) .trim(); if (!bookingId) { range.setValue(''); SpreadsheetApp .getActive() .toast( '找不到 bookingId', '取消失敗', 5 ); return; } /* * 防止多人同時操作 */ const lock = LockService .getScriptLock(); lock.waitLock(15000); try { adminCancelBooking( bookingId ); } finally { try { lock.releaseLock(); } catch (error) {} } } catch (error) { console.error(error); } } /* ================================================== 管理員取消整筆預約 ================================================== */ function adminCancelBooking( bookingId ) { const sheet = getBookingSheet(); const lastRow = sheet.getLastRow(); if (lastRow <= 1) { return; } const values = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); const rowsToDelete = []; const info = { room: '', date: '', name: '', ext: '', times: [] }; values.forEach( (row, index) => { const rowBookingId = String( row[5] || '' ).trim(); if ( rowBookingId !== bookingId ) { return; } rowsToDelete.push( index + 2 ); info.room = String(row[0]); info.date = normalizeDate(row[1]); info.name = String(row[3]); info.ext = String(row[4]); /* * ★ 時間格式修正 */ info.times.push( normalizeTime(row[2]) ); } ); if ( rowsToDelete.length === 0 ) { return; } /* * 取得操作者 */ let operator = ''; try { operator = Session .getActiveUser() .getEmail(); } catch (error) { operator = ''; } if (!operator) { operator = 'Google Sheet 管理員'; } /* * 寫入取消紀錄 */ writeCancelLog({ room: info.room, date: info.date, times: info.times, name: info.name, ext: info.ext, bookingId: bookingId, method: '管理員取消', operator: operator }); /* * 後往前刪除 */ rowsToDelete .sort( (a, b) => b - a ) .forEach( rowNumber => { sheet.deleteRow( rowNumber ); } ); SpreadsheetApp .getActive() .toast( `已取消 ${info.name} 的整筆預約`, '完成', 5 ); } /* ================================================== 寫入取消紀錄 ================================================== */ function writeCancelLog(data) { const sheet = getCancelLogSheet(); const times = [...data.times] .map( time => normalizeTime(time) ) .sort(); const timeRange = formatTimeRange( times ); sheet.appendRow([ new Date(), data.room, data.date, timeRange, data.name, data.ext, data.bookingId, data.method, data.operator ]); } /* ================================================== 把多個小時轉成時間區間 ================================================== */ function formatTimeRange( times ) { if ( !times || times.length === 0 ) { return ''; } const sorted = [...times] .map( time => normalizeTime(time) ) .sort(); const first = sorted[0]; const last = sorted[ sorted.length - 1 ]; const hour = parseInt( last.split(':')[0], 10 ); if (Number.isNaN(hour)) { return first; } const endHour = hour + 1; return ( first + '～' + String(endHour) .padStart(2, '0') + ':00' ); } /* ================================================== 產生取消碼 ================================================== */ function generateCancelCode() { const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let code = ''; for ( let i = 0; i < 6; i++ ) { code += chars.charAt( Math.floor( Math.random() * chars.length ) ); } return code; } /* ================================================== 日期格式 ================================================== */ function normalizeDate( value ) { /* * Google Sheet 日期欄位 */ if ( Object.prototype.toString.call( value ) === '[object Date]' ) { return Utilities.formatDate( value, Session.getScriptTimeZone(), 'yyyy-MM-dd' ); } /* * 已經是文字 */ const text = String(value || '').trim(); /* * yyyy-MM-dd */ if ( /^\d{4}-\d{2}-\d{2}$/.test(text) ) { return text; } /* * yyyy/MM/dd */ const slashMatch = text.match( /^(\d{4})\/(\d{1,2})\/(\d{1,2})$/ ); if (slashMatch) { return ( slashMatch[1] + '-' + String( slashMatch[2] ).padStart(2, '0') + '-' + String( slashMatch[3] ).padStart(2, '0') ); } return text; } /* ================================================== ★ 時間格式 ================================================== */ function normalizeTime( value ) { /* * 情況 1： * Google Sheet 儲存格是 Date 物件 * * 例如： * Sat Dec 30 1899 12:00:00 GMT+0800 * * 轉成： * 12:00 */ if ( Object.prototype.toString.call( value ) === '[object Date]' ) { return Utilities.formatDate( value, Session.getScriptTimeZone(), 'HH:mm' ); } /* * 情況 2： * 已經是文字 */ const text = String(value || '').trim(); if (!text) { return ''; } /* * 已經是 HH:mm */ const simpleMatch = text.match( /^(\d{1,2}):(\d{2})$/ ); if (simpleMatch) { return ( String( simpleMatch[1] ).padStart(2, '0') + ':' + simpleMatch[2] ); } /* * 例如： * * 12:00:00 * 2026/08/20 12:00:00 * Sat Dec 30 1899 12:00:00 GMT+0800 * * 只取其中的時間部分。 */ const match = text.match( /(\d{1,2}):(\d{2})(?::\d{2})?/ ); if (match) { return ( String( match[1] ).padStart(2, '0') + ':' + match[2] ); } return text; } /* ================================================== JSON 回應 ================================================== */ function jsonResponse( data ) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } /* ================================================== 第一次安裝 / 修復設定 ================================================== */ function setupSystem() { const bookingSheet = getBookingSheet(); const logSheet = getCancelLogSheet(); /* * C 欄時間強制設為純文字 */ bookingSheet .getRange( 2, 3, Math.max( bookingSheet.getMaxRows() - 1, 1 ), 1 ) .setNumberFormat('@'); /* * 自動調整欄寬 */ bookingSheet.autoResizeColumns( 1, 8 ); logSheet.autoResizeColumns( 1, CANCEL_LOG_HEADERS.length ); SpreadsheetApp .getActive() .toast( '會議室預約系統設定完成', '完成', 5 ); }
